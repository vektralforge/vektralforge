# Dependencias de ejecución de Airflow — se instalan en la imagen.
#
# La versión de Airflow NO se declara aquí: la fija la etiqueta de la imagen
# base en el Dockerfile. Declararla en dos sitios fue lo que hizo que el
# contenedor corriera 2.9.1 mientras este archivo decía otra cosa.
#
# Los providers van sin fijar versión a propósito: el archivo de constraints de
# Airflow elige la combinación compatible. Fijarlos aquí produce conflictos con
# ese archivo.
#
# Las herramientas de desarrollo (pytest, ruff, detect-secrets) están en
# requirements-dev.txt y no se instalan en la imagen.

# ── Linaje ───────────────────────────────────────────────────────────────────
apache-airflow-providers-openlineage

# ── Autenticación (FabAuthManager: usuarios y roles) ─────────────────────────
apache-airflow-providers-fab

# ── Motores de procesamiento y consulta ──────────────────────────────────────
apache-airflow-providers-apache-spark
apache-airflow-providers-apache-hive
apache-airflow-providers-apache-kafka

# ── Conectores de origen ─────────────────────────────────────────────────────
apache-airflow-providers-microsoft-mssql

# ── Cliente HTTP ─────────────────────────────────────────────────────────────
# Los DAGs y airflow/plugins/http_publico.py importan requests y urllib3
# directamente. Llegaban como transitivas de Airflow, que funciona hasta que una
# versión deja de arrastrarlas. Sin fijar versión, por el mismo motivo que los
# providers: la elige el archivo de constraints. urllib3 viene con requests.
requests

# ── DAGs desde Git ───────────────────────────────────────────────────────────
# Trae el GitDagBundle: en Kubernetes los DAGs y sus jobs de Spark se clonan del
# repositorio en vez de montarse desde el host, y cada DAG run queda atado al
# commit con que se creó. La imagen base ya lo trae (extra «git»); se declara
# por lo mismo que requests. La versión la fija el constraints: 0.4.0 en 3.3.0.
apache-airflow-providers-git

# ── Secretos ─────────────────────────────────────────────────────────────────
hvac

# ── Logs de tareas en el object store ────────────────────────────────────────
# Trae el S3TaskHandler: los logs de cada tarea se suben al bucket airflow-logs
# y la UI los lee de ahí, sin un volumen compartido entre los pods de Airflow.
# Ver [logging] en docker-compose.yml.
#
# Aquí había una nota que lo dejaba fuera por «incompatibilidad con SQLAlchemy
# 2.x». Ya no aplica: el archivo de constraints de Airflow 3.3.0 fija
# apache-airflow-providers-amazon==9.31.0 junto a SQLAlchemy==2.0.51, y la imagen
# base oficial lo trae instalado en sus extras por defecto. Se declara igual,
# por lo mismo que requests: lo que usa el proyecto se nombra, no se hereda.
apache-airflow-providers-amazon

# ── Cliente Spark ────────────────────────────────────────────────────────────
# SparkSubmitOperator ejecuta spark-submit desde el contenedor de Airflow, no
# desde el de Spark. Requiere pyspark (que provee el binario) y una JVM, que se
# instala en el Dockerfile.
#
# La versión debe coincidir con la de spark-master y spark-worker: un cliente y
# un cluster desalineados fallan con errores de protocolo poco descriptivos.
pyspark==4.1.3
delta-spark==4.1.0
